{
    std::string err;
    if (!m_objCatalog.AppendFromFile(
            (pathutils::ProjectRoot() / "assets" / "Objects" / "Smithy" / "SmithyObjects.json").string(),
            &err))
    {
        SDL_ShowSimpleMessageBox(
            SDL_MESSAGEBOX_ERROR,
            "Objects JSON error (Smithy)",
            err.c_str(),
            m_window);
        return false;
    }
}
